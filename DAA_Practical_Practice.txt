class Node:
	def __init__(self, char=None, freq=None, left=None, right= None):
		self.char = char
		self.freq = freq
		self.left = left
		self.right = right

	def __lt__(self,other):
		return self.freq < other.freq

def Huffman(text):

	frequencies = Counter(text)
	
	if not frequencies:
		return {}


	heap = [Node(char,freq) for char, freq in frequencies.items()]
	
	heapq.heapify(heap)
	
	if len(heap) == 1:
		return {heap[0].char:"0"}


	while(len(heap) > 1):
		
		right = heapq.heappop(heap)
		left = heapq.heappop(heap)
		
		parent = heapq.push(Node(freq = right.freq + left.freq, left = left, right = right))

		heapq.heappush(heap, parent)

	codes = {}

	def generate_code(node, code = ""):
		
		if node.char is not None:
			codes[node.char]= code
		else:
			generate_code(node.left, code+"0")	
			generate_code(node.right, code+"1")


	generate_code(heap[0])


	return codes

def fractional_knapsack(weights, profits, capacity, show_items=False):
    items = list(zip(weights, profits))

    ratios = []

    for item in items:
        ratio = item[1] / item[0]
        ratios.append(ratio)

    items = list(zip(ratios, items))
    items.sort(reverse=True)

    total_profit = 0
    selected_items = []

    for ratio, item in items:

        weight = item[0]
        profit = item[1]

        if capacity <= 0:
            break

        take = min(weight, capacity)

        total_profit += take * (profit / weight)

        selected_items.append((weight, profit, take))

        capacity -= take

    if show_items:
        print("\nItems selected:")

        for weight, profit, take in selected_items:
            print("Weight:", weight,
                  "Profit:", profit,
                  "Weight Taken:", take)

    return total_profit

def dp_knapsack(weights, profits, capacity):
	
	n = len(weights)
	
	dp = [[0]* (capacity+1) for i in range(n+1)]
	

	for i in range(1, n + 1):
		for w in range(capacity + 1):
			
			dp[i][w] = dp[i-1][w]
			
			if weights[i-1] <= w:
			
				dp[i][w] = max(dp[i-1][w], profits[i-1] + dp[i-1][w - weights[i-1]]) 	


	return dp[n][capacity]

def solve_n_queens(n, fixed_position = None):

	board = [-1]*n
	solutions = []
	cols, pos_diag, neg_diag = set(), set(), set()

	if fixed_position is not None:
		r, c = fixed_position
		
		board[r] = c
	
		cols.add(c)
		
		pos_diag.add(r+c)

		neg_diag.add(r-c)

	
	def backtrack(row):
		
		if row == n:
			solutions.append(list(board))
			return
		
		if fixed_position is not None and row == r:
			backtrack(row+1)
			return 
		
		for col in range(n):
			
			if col in cols or (row + col) in pos_diag or (row-col) in neg_diag:
				continue
			
			
			board[row] = col
			
			cols.add(col)
			pos_diag.add(row+col)
			neg_diag.add(row-col)



			backtrack (row+1)


			board[row] = -1
			cols.remove(col)
			pos_diag.add(row+col)
			neg_diag.add(row-col)

	backtrack(0)
	return solutions
